kind: ClusterRole
metadata:
  #namespace: test ## 所屬的名稱空間
  name: ns-reader
rules: ## 當前角色的規則
- apiGroups: [""] # "" 標明 core API 組
  resources: ["namespaces"]
  verbs: ["get", "watch", "list"] # 動詞。
---
## 编写角色和账号的绑定关系
apiVersion: rbac.authorization.k8s.io/v1
# 此角色绑定允许 "jane" 读取 "default" 名称空间中的 Pods
kind: ClusterRoleBinding
metadata:
  name: ns-reader
subjects:
# 你可以指定不止一个“subject（主体）”
- kind: ServiceAccount
  name: km # "name" 是不区分大小写的
  namespace: test
  apiGroup: rbac.authorization.k8s.io
roleRef:
  # "roleRef" 指定与某 Role 或 ClusterRole 的绑定关系
  kind: ClusterRole # 此字段必须是 Role 或 ClusterRole
  name: pod-reader      # 此字段必须与你要绑定的 Role 或 ClusterRole 的名称匹配
  apiGroup: rbac.authorization.k8s.io
---
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: read-ns-global
subjects:
- kind: ServiceAccount
  name: km # 'name' 是不区分大小写的
  namespace: test
  apiGroup: rbac.authorization.k8s.io
roleRef:
  kind: ClusterRole
  name: ns-reader
